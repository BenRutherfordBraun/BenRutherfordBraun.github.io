#include "RoverState.h"
#include "MotorController.h"
#include "VisionSystem.h"

void updateRoverState(RoverState& currentState, const VisionSystem& vision, MotorController& motors) {
    switch (currentState) {
        case RoverState::SEARCHING:
            motors.executeSearchPattern();
            if (vision.targetDetected()) {
                currentState = RoverState::TRACKING;
            }
            break;
            
        case RoverState::TRACKING:
            motors.trackTarget(vision.getTargetHeading());
            if (vision.inGraspRange()) {
                motors.stop();
                currentState = RoverState::GRASPING;
            }
            break;
            
        case RoverState::FAULT:
            motors.emergencyStop();
            break;
    }
}
#include "MotorController.h"
#include <Arduino.h>

MotorController::MotorController(uint8_t leftPin, uint8_t rightPin) 
    : leftMotorPwm(leftPin), rightMotorPwm(rightPin) {}

void MotorController::trackTarget(float headingError) {
    // Basic Proportional control for steering
    float kp = 1.5f;
    float steeringAdjust = headingError * kp;
    
    float leftSpeed = baseSpeed + steeringAdjust;
    float rightSpeed = baseSpeed - steeringAdjust;
    
    setSpeeds(constrain(leftSpeed, -255, 255), constrain(rightSpeed, -255, 255));
}

void MotorController::emergencyStop() {
    setSpeeds(0, 0);
    // Engage hardware brakes if applicable
}
